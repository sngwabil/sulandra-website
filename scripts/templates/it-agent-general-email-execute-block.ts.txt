      else if(audience==='HR_ADMIN'){query+=' AND "role"::text=ANY($2::text[])';args.push(hrRoles)}

          : `No intended recipient was confirmed accepted by SMTP. This email must not be described as sent or delivered.`;

        smtpAccepted:delivery.smtpAcceptedCount>0,
